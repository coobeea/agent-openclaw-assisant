```python
        pass

        """
        pass
    
    def release_container(container_id):
        """释放容器回池子"""
        pass
    
    # 4. 对话管理
    def send_message(agent_id, message):
        """
        发送消息给智能体
        1. 分配容器
        2. 加载智能体数据
        3. 执行对话
        4. 保存结果
        5. 释放容器
        """
        pass
    
    def get_history(agent_id):
        """获取对话历史"""
        pass
```

**技术栈选择**：
- **Python + Flask/FastAPI**（推荐）：简单、快速
- **Node.js + Express**：和 OpenClaw 同语言

**部署方式**：
```yaml
# 05-scheduler-service.yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: scheduler-service
  namespace: openclaw
spec:
  replicas: 1
  template:
    spec:
      containers:
      - name: scheduler
        image: scheduler:latest
        ports:
        - containerPort: 8080
        env:
        - name: POSTGRES_HOST
          value: "postgres"
        - name: MINIO_ENDPOINT
          value: "minio:9000"
```
